\begin{frame}{Problem Statement}
\begin{itemize}
  \item No trainer supervision during home workouts
  \item Wrong form increases injury risk
  \item Manual rep counting is distracting
  \item Existing apps mostly detect \textbf{what} exercise is happening
  \item FormFix AI focuses on \textbf{how well} the exercise is performed
\end{itemize}
\end{frame}

\begin{frame}{How Accuracy Is Calculated}
\small
\textbf{Top-1 Accuracy}
\[
\text{Top-1}=\frac{\text{Correct highest-confidence predictions}}{\text{Total predictions}}\times 100
\]

\textbf{Top-3 Accuracy}
\[
\text{Top-3}=\frac{\text{True class in top 3 predictions}}{\text{Total predictions}}\times 100
\]

\textbf{Precision / Recall / F1}
\[
\text{Precision}=\frac{TP}{TP+FP}
\quad
\text{Recall}=\frac{TP}{TP+FN}
\quad
\text{F1}=2\cdot\frac{PR}{P+R}
\]
\end{frame}
