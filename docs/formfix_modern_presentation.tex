\begin{frame}{Model Strategy: Why Three Models?}
    \begin{table}
        \centering
        \small
        \begin{tabular}{p{2.2cm} p{3.2cm} p{5.1cm}}
            \toprule
            \textbf{Model} & \textbf{Role} & \textbf{Reason for Use} \\
            \midrule
            LSTM & Baseline model & Lightweight and fast sequence model for comparison. \\
            BiLSTM & Main temporal model & Uses forward and backward context; stronger motion understanding. \\
            Transformer & Advanced comparison & Uses self-attention for long-range frame relationships. \\
            \bottomrule
        \end{tabular}
    \end{table}

    \vspace{0.3cm}
    \centering
    \textbf{Deployment logic:} the system supports \textit{Fast}, \textit{Best}, \textit{Hybrid}, and \textit{Ensemble} modes.\\
    In practice, BiLSTM is weighted highest in the hybrid pipeline.
\end{frame}

\begin{frame}{How Accuracy Is Calculated}
    \small
    \textbf{Top-1 Accuracy}
    \[
    \text{Top-1} = \frac{\text{Correct highest-confidence predictions}}{\text{Total predictions}} \times 100
    \]

    \textbf{Top-3 Accuracy}
    \[
    \text{Top-3} = \frac{\text{True class inside top 3 predictions}}{\text{Total predictions}} \times 100
    \]

    \textbf{Precision, Recall, F1-score}
    \[
    \text{Precision}=\frac{TP}{TP+FP},\quad
    \text{Recall}=\frac{TP}{TP+FN},\quad
    \text{F1}=2\cdot\frac{PR}{P+R}
    \]
